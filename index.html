Cole o codigo abaixo
Troque "Inventario" pelo nome exato da sua aba.
function doPost(e) { try { var sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName("Inventario"); var dados = JSON.parse(e.postData.contents); dados.forEach(function(item) { sheet.appendRow([ item.numero, item.data, item.hora ]); }); return ContentService .createTextOutput("OK") .setMimeType(ContentService.MimeType.TEXT); } catch(err) { return ContentService .createTextOutput("ERRO: " + err) .setMimeType(ContentService.MimeType.TEXT); } }
